-- ── Auto Steal ───────────────────────────────────────────────
BtnAutoSteal.MouseButton1Click:Connect(function()
    featureStates.AutoSteal = not featureStates.AutoSteal

    if featureStates.AutoSteal then
        ActivateButton(BtnAutoSteal, StrokeAutoSteal, DotAutoSteal)

        -- Turn off Auto Bat if it was active (mutual exclusion)
        featureStates.AutoBat = false
        DeactivateButton(BtnAutoBat, StrokeAutoBat, DotAutoBat)

        Heartbeat:Connect(function()
            if not featureStates.AutoSteal then return end
            local char = LocalPlayer.Character
            if not char then return end
            -- Iterate all other players and attempt to steal within radius
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local otherRoot = player.Character:FindFirstChild("HumanoidRootPart")
                    local myRoot    = char:FindFirstChild("HumanoidRootPart")
                    if otherRoot and myRoot then
                        local dist = (myRoot.Position - otherRoot.Position).Magnitude
                        if dist <= STEAL_RADIUS then
                            -- Fire steal remote (game-specific, adapt as needed)
                            local remote = game:GetService("ReplicatedStorage"):FindFirstChild("StealBall")
                            if remote then
                                remote:FireServer(player)
                            end
                        end
                    end
                end
            end
        end)
    else
        DeactivateButton(BtnAutoSteal, StrokeAutoSteal, DotAutoSteal)
    end
end)

-- ── Auto Bat + Fling (share toggle) ──────────────────────────
BtnAutoBat.MouseButton1Click:Connect(function()
    featureStates.AutoBat = not featureStates.AutoBat

    if featureStates.AutoBat then
        ActivateButton(BtnAutoBat, StrokeAutoBat, DotAutoBat)

        -- Also visually activate Fling button (they work together in the log)
        ActivateButton(BtnFling, StrokeFling, DotFling)
        featureStates.Fling = true

        -- AutoBat: move character toward ball on every Heartbeat
        AutoBatConnection = Heartbeat:Connect(function()
            if not featureStates.AutoBat then return end
            local char = LocalPlayer.Character
            if not char then return end
            -- Game-specific: find the ball and move toward it
            local ball = workspace:FindFirstChild("Ball") or workspace:FindFirstChild("SoccerBall")
            if ball then
                local root = char:FindFirstChild("HumanoidRootPart")
                if root then
                    root.CFrame = CFrame.new(root.Position, ball.Position)
                end
            end
        end)

        -- Fling: apply high velocity on RenderStepped
        AutoBatRenderConnection = RenderStepped:Connect(function()
            if not featureStates.Fling then return end
            local char = LocalPlayer.Character
            if not char then return end
            local root = char:FindFirstChild("HumanoidRootPart")
            if root then
                local ball = workspace:FindFirstChild("Ball")
                if ball and (root.Position - ball.Position).Magnitude < 5 then
                    ball.AssemblyLinearVelocity = root.CFrame.LookVector * 200
                end
            end
        end)
    else
        ActivateButton(BtnAutoBat, StrokeAutoBat, DotAutoBat)   -- toggle to inactive
        DeactivateButton(BtnAutoBat, StrokeAutoBat, DotAutoBat)
        DeactivateButton(BtnFling, StrokeFling, DotFling)
        featureStates.Fling = false
        if AutoBatConnection then AutoBatConnection:Disconnect() end
        if AutoBatRenderConnection then AutoBatRenderConnection:Disconnect() end
    end
end)

            -- Wait for the character (or current character respawn)
            local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
            local root = char:WaitForChild("HumanoidRootPart")

            Heartbeat:Connect(function()
                if not featureStates.AutoPlayRight then return end
                if not root or not root.Parent then return end

                local targetPos = Vector3.new(POS_RIGHT.X, root.Position.Y, POS_RIGHT.Z)
                local direction = targetPos - root.Position

                if direction.Magnitude > 2 then
                    -- Velocity-based movement (matches log: AssemblyLinearVelocity)
                    root.AssemblyLinearVelocity = Vector3.new(
                        direction.Unit.X * AUTO_PLAY_SPEED,
                        root.AssemblyLinearVelocity.Y,
                        direction.Unit.Z * AUTO_PLAY_SPEED
                    )
                else
                    root.AssemblyLinearVelocity = Vector3.new(0, root.AssemblyLinearVelocity.Y, 0)
                end
            end)
        end)
    else
        BtnAutoPlayRight.BackgroundTransparency = 0.85
        BtnAutoPlayRight.TextColor3             = Color3.fromRGB(255, 255, 255)
        StrokeAutoPlayRight.Color               = Color3.fromRGB(200, 200, 200)
        StrokeAutoPlayRight.Transparency        = 0.4
    end
end)

-- ── Character respawn handler ─────────────────────────────────
LocalPlayer.CharacterAdded:Connect(function(character)
    task.wait(1)
    RefreshCharacter()

    -- Disconnect previous per-character connections
    for _, conn in ipairs(connections) do
        if typeof(conn) == "RBXScriptConnection" then
            conn:Disconnect()
        end
    end
    connections = {}

    task.wait(0.1)

    -- Re-hook AutoBat if it was active
    if featureStates.AutoBat and AutoBatConnection then
        AutoBatConnection:Disconnect()
        AutoBatConnection = nil
    end

    -- Re-hook RenderStepped FreezePlayer-style anti-ragdoll
    table.insert(connections, RunService.Stepped:Connect(function()
        local char = LocalPlayer.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
            hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
        end
    end))

    task.wait(0.3)

    -- Wait for humanoid, then add Died connection
    local humanoid = character:FindFirstChildWhichIsA("Humanoid")
    if humanoid then
        table.insert(connections, humanoid.Died:Connect(function()
            -- Clean up on death
            for _, conn in ipairs(connections) do
                if typeof(conn) == "RBXScriptConnection" then
                    conn:Disconnect()
                end
            end
            connections = {}
        end))
    end

    task.wait(0.5)
    RefreshCharacter()
end)
